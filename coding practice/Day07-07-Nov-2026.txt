 🔹 Ansible - setup Nginx with custom config

 name: Setup Ngnix with custom config hosts: all become:yes tasks:
 name: Install ngnix yum:name:ngnix state:present
 
 name:Copy custom ngnix.conf copy:src ngnix.conf dest:/etc/ngnix/ngnix.conf

 name: Start ngnix service
 service: name: ngnix state:started enabled:yes

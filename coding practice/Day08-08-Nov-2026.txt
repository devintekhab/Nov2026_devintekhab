 🔹jenkins - Docker build and push pipeline
 {
    agent any stage
     {
        stage("Build docker image")
        steps {
            script {docker.build("devintekhab-app")}
        }
     }
     stage("Push docker image")
     {
        steps {
            script{
                docker.withRegistery('https://docker.io','docker-credentials){
                    docker.image('devintekhab-app').push('latest')
                }
            }
        }
     }
 }

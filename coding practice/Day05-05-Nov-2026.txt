🔹 Ansible - Create user and set permission
-name: Create a new user
hosts: all
become: yes
tasks:
   -name: create user
   user:
     name:devintekhab
     state:present
     shell:/bin/bash
    -name:Set Permissions for the user
    file:
       path:/home/devintekhab
       owner:devintekhab
       group:devintekhab
       mode:'0755'

🔹 Ansible - create docker containers for multiples application

- name: setup docker containers for multiple apps
  hosts: all
  become: yes
  tasks:
    - name: Pull web app image
      docker_image:
          name: my-web-app
          source: pull


    - name: pull database app image
      docker_image: 
          name: my-database-app
          source: pull

    - name: run database app container
      docker_container: 
         name: database-app
         image: my database-app
         state: started
         published_ports:
         -  "3306:3306"
  
🔹Terraform - vpc with private and public subnets

    cidr_block = '10.0.0.0/16'
}

resources "aws_subnet" "public" {
     vpc_id            = aws_vpc.main.id

}

🔹 Kubernets - deploying statefulset
apiVersion: apps/v1
kind: statefulSets
metadata:
    name: devintekhab-db
spec:  
  serviceName: "devintekhab-db"
  replicas: 3
  selector:
     matchlabels:
         app: devintekhab-app
  template:
     metadata:
         labels:
           apps: devintekhab-app
     spec:
       containers:
          - name: db
            image: mysql:5.7
            volumeMounts:
                - mountPath: /var/lib/mysql
                  name: mysql-storage


    volumeClaimsTemplates:
        - metadata: 
            name: mysql-storage
          spec:
           accessModes: ["ReadwriteOnes"]
           resources:
             requests:
                storage: 1Gi


🔹 Jenkinsfile - build annd deploy to kubernetics
pipeline {
    agent any
    enviroment{
        IMAGE_NAME: "devintekhab_web_app"  
    }
    stages {
        stage('build docker image'){
            steps {
                script {
                    docker.build("$env.IMAGE_NAME")
                }
            }
        }
    }
 
 stage('deploy docker image'){
    steps{
        script {
            docker.withRegistry('https://docker.io','docker-credentials'){
                docker.image('$env.IMAGE_NAME).push("latest")
            }
        }
    }
 }

stage("deploy to kubernetics"){
    steps {
        script {
            sh 'kubectl apply -f deployment.yaml'
        }
    }
}
}
🔹 gitlab CI/CD build,push and deploy
stages:
   - build
   - push
   - deploy

build:
   stage:build
   script:
     - docker build -t devintekhab-web-app

push: 
   stage: push
   script:
     - docker login -u '$CIREGISTERY_USER" -p '$CI_REGISTERY_PASSWORD'  
     - docker push devintekhab-web-app
  

deploy: 
  stage: deploy
  script:
   - kubectl apply -f kubernetes/deployment.yaml
